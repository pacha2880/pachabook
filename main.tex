% Página completa con una imagen (tapa/contratapa). Las imágenes traen
% 3 mm de sangrado por lado (11.4 x 14.55 cm): lo que sobra queda fuera de la página.
\newcommand{\paginaimagen}[1]{%
  \clearpage
  \thispagestyle{empty}%
  \AddToShipoutPictureBG*{\AtPageCenter{%
    \makebox(0,0){\includegraphics[width=11.4cm]{#1}}}}%
  \mbox{}%
  \clearpage
}

\newcommand{\paginablanco}{%
  \clearpage
  \thispagestyle{empty}%
  \mbox{}%
  \clearpage
}
