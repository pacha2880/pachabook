% Página completa con una imagen (tapa/contratapa). Las imágenes miden
% exactamente la página (10.8 x 13.95 cm, sin sangrado).
% Si \sintapas está definido (ver pachabook-sin-tapas.tex), no genera nada:
% sale el libro sin tapa ni contratapa, conservando las páginas en blanco.
\newcommand{\paginaimagen}[1]{%
  \ifdefined\sintapas\else
    \clearpage
    \thispagestyle{empty}%
    \AddToShipoutPictureBG*{\AtPageCenter{%
      \makebox(0,0){\includegraphics[width=\paperwidth,height=\paperheight]{#1}}}}%
    \mbox{}%
    \clearpage
  \fi
}

\newcommand{\paginablanco}{%
  \clearpage
  \thispagestyle{empty}%
  \mbox{}%
  \clearpage
}

% ================== TAPA (sin número) ==================
\pagenumbering{gobble}    % oculta numeración antes del contenido
